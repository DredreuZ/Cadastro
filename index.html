    // --- GERADOR DE EAN-13 EM SVG (PURE JS, SEM DEPENDﾃ劾CIAS EXTERNAS) ---

    const EAN13_PATTERNS = {

      L: ["0001101", "0011001", "0010011", "0111101", "0100011", "0110001", "0101111", "0111011", "0110111", "0001011"],

      G: ["0100111", "0110011", "0011011", "0100001", "0011101", "0111001", "0000101", "0010001", "0001001", "0010111"],

      R: ["1110010", "1100110", "1101100", "1000010", "1011000", "1001110", "1010000", "1000100", "1001000", "1110100"]

    };

    const PARITY_MAP = [

      "LLLLLL", "LLGLGG", "LLGGLG", "LLGGGL", "LGLLGG",

      "LGGLLG", "LGGGLL", "LGLGLG", "LGLGGL", "LGGLGL"

    ];

    function gerarSVGBarraEAN13(codigo) {

      // Normaliza e garante 13 dﾃｭgitos numﾃｩricos

      let clean = codigo.replace(/\D/g, '');

      if (clean.length < 13) clean = clean.padEnd(13, '0');

      clean = clean.substring(0, 13);

      const firstDigit = parseInt(clean[0]);

      const parityPattern = PARITY_MAP[firstDigit];

      let bits = "101"; // Left guard

      // Primeiros 6 dﾃｭgitos (esquerda)

      for (let i = 0; i < 6; i++) {

        const digit = parseInt(clean[i + 1]);

        const type = parityPattern[i];

        bits += EAN13_PATTERNS[type][digit];

      }

      bits += "01010"; // Center guard

      // ﾃ嗟timos 6 dﾃｭgitos (direita)

      for (let i = 0; i < 6; i++) {

        const digit = parseInt(clean[i + 7]);

        bits += EAN13_PATTERNS['R'][digit];

      }

      bits += "101"; // Right guard

      // Monta as barras em SVG

      const barWidth = 1.6;

      const height = 40;

      const fontHeight = 11;

      const totalWidth = bits.length * barWidth + 16;

      const totalHeight = height + fontHeight + 2;

      let rects = '';

      for (let i = 0; i < bits.length; i++) {

        if (bits[i] === '1') {

          const x = 8 + i * barWidth;

          // Guard bars extend slightly lower

          const isGuard = (i < 3) || (i >= 45 && i <= 49) || (i >= 92);

          const barH = isGuard ? height + 4 : height;

          rects += `<rect x="${x.toFixed(1)}" y="0" width="${barWidth}" height="${barH}" fill="#000" />`;

        }

      }

      const svg = `

        <svg class="barcode-svg" viewBox="0 0 ${totalWidth} ${totalHeight}" xmlns="http://www.w3.org/2000/svg">

          <g>${rects}</g>

          <text x="${totalWidth / 2}" y="${height + fontHeight}" font-family="monospace" font-size="10" font-weight="bold" text-anchor="middle" fill="#000">

            ${clean[0]} ${clean.substring(1,7)} ${clean.substring(7,13)}

          </text>

        </svg>

      `;

      return svg;

    }
